CELL 2 — MODEL

import torch
from unsloth import FastLanguageModel

assert torch.cuda.is_available(), "CUDA GPU required"

gpu_memory_gb = (
    torch.cuda.get_device_properties(0).total_memory
    / (1024 ** 3)
)

dtype = (
    torch.bfloat16
    if torch.cuda.is_bf16_supported()
    else torch.float16
)

max_seq_length = 3072

model, tokenizer = FastLanguageModel.from_pretrained(
    model_name="Qwen/Qwen2.5-Coder-1.5B-Instruct",
    max_seq_length=max_seq_length,
    dtype=dtype,
    load_in_4bit=True,
)

if tokenizer.pad_token is None:
    tokenizer.pad_token = tokenizer.eos_token

tokenizer.padding_side = "right"

model = FastLanguageModel.get_peft_model(
    model,
    r=64,
    lora_alpha=128,
    lora_dropout=0.05,
    bias="none",
    target_modules=[
        "q_proj",
        "k_proj",
        "v_proj",
        "o_proj",
        "gate_proj",
        "up_proj",
        "down_proj",
    ],
    use_gradient_checkpointing="unsloth",
    random_state=3407,
)

model.print_trainable_parameters()

CELL 3 — DATASET

from datasets import load_dataset, Dataset, concatenate_datasets
import hashlib

SYSTEM_PROMPT = """You are Spark, Catalyst's fast local coding assistant.
Solve small coding tasks, syntax fixes, scripts, and concise explanations.
Do not invent files, APIs, tests, or tool output.
Never claim unrun tests passed.
Use fenced code and be direct.
"""


def get_value(row, *names):
    for name in names:
        value = row.get(name)
        if value is not None and str(value).strip():
            return str(value).strip()
    return ""


def normalize_row(row):
    prompt = get_value(
        row,
        "instruction",
        "prompt",
        "question",
        "problem",
    )
    context = get_value(
        row,
        "input",
        "context",
        "additional_context",
    )
    answer = get_value(
        row,
        "output",
        "response",
        "completion",
        "answer",
    )

    if context:
        prompt += f"\n\nAdditional context:\n{context}"

    return {
        "prompt": [
            {"role": "system", "content": SYSTEM_PROMPT},
            {"role": "user", "content": prompt},
        ],
        "completion": [
            {"role": "assistant", "content": answer},
        ],
    }


raw = load_dataset(
    "sahil2801/CodeAlpaca-20k",
    split="train",
)

data = raw.map(
    normalize_row,
    remove_columns=raw.column_names,
)

data = data.filter(
    lambda row: (
        len(row["prompt"][1]["content"]) >= 12
        and 20 <= len(row["completion"][0]["content"]) <= 12000
    )
)

# Small targeted examples teach Spark Catalyst-specific behavior.
# Repeat the set a few times, but never repeat only one example hundreds of times.
synthetic_examples = [
    {
        "prompt": [
            {"role": "system", "content": SYSTEM_PROMPT},
            {
                "role": "user",
                "content": (
                    "Fix this Python function:\n\n"
                    "```python\n"
                    "def add(a, b)\n"
                    "    return a + b\n"
                    "```"
                ),
            },
        ],
        "completion": [
            {
                "role": "assistant",
                "content": (
                    "The function is missing a colon after the parameter list.\n\n"
                    "```python\n"
                    "def add(a, b):\n"
                    "    return a + b\n"
                    "```"
                ),
            }
        ],
    },
    {
        "prompt": [
            {"role": "system", "content": SYSTEM_PROMPT},
            {"role": "user", "content": "Write a JavaScript debounce helper."},
        ],
        "completion": [
            {
                "role": "assistant",
                "content": (
                    "```javascript\n"
                    "function debounce(fn, delay) {\n"
                    "  let timer;\n\n"
                    "  return (...args) => {\n"
                    "    clearTimeout(timer);\n"
                    "    timer = setTimeout(() => fn(...args), delay);\n"
                    "  };\n"
                    "}\n"
                    "```"
                ),
            }
        ],
    },
    {
        "prompt": [
            {"role": "system", "content": SYSTEM_PROMPT},
            {
                "role": "user",
                "content": "Can you claim that my code works without running it?",
            },
        ],
        "completion": [
            {
                "role": "assistant",
                "content": (
                    "No. I can inspect the code and identify likely problems, "
                    "but I should not claim it passed unless it was actually "
                    "executed or tested."
                ),
            }
        ],
    },
    {
        "prompt": [
            {"role": "system", "content": SYSTEM_PROMPT},
            {
                "role": "user",
                "content": "What should I inspect before changing a Node.js project?",
            },
        ],
        "completion": [
            {
                "role": "assistant",
                "content": (
                    "Start with package.json, the source entry point, configuration "
                    "files, existing tests, and the modules imported by the code "
                    "you plan to change. Inspect the smallest relevant set first."
                ),
            }
        ],
    },
]

synthetic = Dataset.from_list(synthetic_examples)
synthetic = concatenate_datasets([synthetic] * 6)
data = concatenate_datasets([data, synthetic])
data = data.shuffle(seed=3407)


def add_hash(row):
    content = (
        row["prompt"][1]["content"]
        + "\n"
        + row["completion"][0]["content"]
    )
    return {
        "example_hash": hashlib.sha256(
            content.encode("utf-8")
        ).hexdigest()
    }


data = data.map(add_hash)
seen = set()


def keep_unique(row):
    key = row["example_hash"]
    if key in seen:
        return False
    seen.add(key)
    return True


data = data.filter(keep_unique)
data = data.remove_columns("example_hash")

split = data.train_test_split(
    test_size=0.05,
    seed=3407,
)

train_dataset = split["train"]
eval_dataset = split["test"]

print(f"Train: {len(train_dataset)}")
print(f"Eval:  {len(eval_dataset)}")

CELL 4 — TRAIN

from trl import SFTTrainer, SFTConfig

training_args = SFTConfig(
    output_dir="spark_1.5b_outputs",
    num_train_epochs=2,
    per_device_train_batch_size=(
        4 if gpu_memory_gb >= 30 else 1
    ),
    per_device_eval_batch_size=1,
    gradient_accumulation_steps=8,
    learning_rate=1e-4,
    lr_scheduler_type="cosine",
    warmup_ratio=0.05,
    weight_decay=0.01,
    optim="adamw_8bit",
    max_grad_norm=1.0,
    bf16=(dtype == torch.bfloat16),
    fp16=(dtype == torch.float16),
    max_length=max_seq_length,
    packing=True,
    completion_only_loss=True,
    eval_strategy="steps",
    eval_steps=100,
    save_strategy="steps",
    save_steps=100,
    save_total_limit=3,
    load_best_model_at_end=True,
    metric_for_best_model="eval_loss",
    greater_is_better=False,
    logging_steps=10,
    report_to="none",
    seed=3407,
    data_seed=3407,
)

trainer = SFTTrainer(
    model=model,
    processing_class=tokenizer,
    train_dataset=train_dataset,
    eval_dataset=eval_dataset,
    args=training_args,
)

train_result = trainer.train()
print(train_result.metrics)

eval_metrics = trainer.evaluate()
print(eval_metrics)
