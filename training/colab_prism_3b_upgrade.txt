from datasets import load_dataset
import hashlib
SYSTEM_PROMPT="You are Prism, Catalyst's reliable local coding model. Handle multi-file coding, debugging, scripts, and implementation. Ground claims in evidence; never invent files, APIs, tests, or tool output. Never claim unrun tests passed. Use fenced code and state assumptions briefly."
def val(r,*ns):
    for n in ns:
        if r.get(n) is not None and str(r[n]).strip(): return str(r[n]).strip()
    return ""
def norm(r):
    p=val(r,"instruction","prompt","question","problem"); c=val(r,"input","context","additional_context"); a=val(r,"output","response","completion","answer")
    if c: p += "\n\nAdditional context:\n"+c
    return {"prompt":[{"role":"system","content":SYSTEM_PROMPT},{"role":"user","content":p}],"completion":[{"role":"assistant","content":a}]}
raw=load_dataset("sahil2801/CodeAlpaca-20k",split="train")
data=raw.map(norm,remove_columns=raw.column_names)
data=data.filter(lambda r:len(r["prompt"][1]["content"])>=16 and 30<=len(r["completion"][0]["content"])<=16000)
data=data.map(lambda r:{"example_hash":hashlib.sha256((r["prompt"][1]["content"]+"\n"+r["completion"][0]["content"]).encode()).hexdigest()})
seen=set()
def keep_unique(r):
    key=r["example_hash"]
    if key in seen: return False
    seen.add(key); return True
data=data.filter(keep_unique).remove_columns("example_hash")
split=data.train_test_split(test_size=.05,seed=3407); train_dataset=split["train"]; eval_dataset=split["test"]

trainer=SFTTrainer(model=model,processing_class=tokenizer,train_dataset=train_dataset,eval_dataset=eval_dataset,args=args)
print(trainer.train().metrics); print(trainer.evaluate())
